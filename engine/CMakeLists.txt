cmake_minimum_required(VERSION 3.16)
project(money_graph_engine VERSION 1.2.0 LANGUAGES CXX)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/default.json" DEFAULT_CONFIG_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS config/default.json)
configure_file(src/default_config.hpp.in generated/default_config.hpp @ONLY)
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
add_executable(engine src/main.cpp src/analysis.cpp)
target_compile_features(engine PRIVATE cxx_std_17)
set_target_properties(engine PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(engine PRIVATE third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(engine PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(engine PRIVATE -static -static-libgcc -static-libstdc++ -municode)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME engine_contract_and_rules
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_engine.py" "$<TARGET_FILE:engine>")
  endif()
endif()
